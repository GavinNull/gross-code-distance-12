% =====================================================================
% PREVIEW WRAPPER -- not the submission file.
% revtex4-2 is not available offline, so this wrapper builds a preview of
% paper/draft.tex with the standard article class.  It swallows draft.tex's
% \documentclass line, provides minimal stand-ins for the revtex-only
% commands/environments used there, and \input's the whole file.
% Layout (one column, float placement, bibliography style) is NOT the
% revtex/PRL layout; use only to proof-read text, tables, figures, refs.
% Build (offline, cached bundle only):
%   cd paper && tectonic --only-cached preview.tex
% =====================================================================
\documentclass{article}
\usepackage[margin=2.2cm]{geometry}
\makeatletter
% swallow the \documentclass of draft.tex
\renewcommand{\documentclass}[2][]{}
% packages of draft.tex that are not in the offline cache: mark as loaded and
% provide the few commands the draft uses from them
\@namedef{ver@amssymb.sty}{}\@namedef{ver@amsfonts.sty}{}\@namedef{ver@hyperref.sty}{}\@namedef{opt@hyperref.sty}{colorlinks=true,linkcolor=blue,citecolor=blue,urlcolor=blue}
\providecommand{\mathbb}[1]{\mathbf{#1}}
\providecommand{\square}{\mathord{\fboxsep=0pt\fbox{\rule{0pt}{1.2ex}\hspace{1.2ex}}}}
\providecommand{\gtrsim}{\mathrel{\vcenter{\offinterlineskip\hbox{$>$}\kern0.2ex\hbox{$\sim$}}}}
% fonts: only TeX Gyre Termes X (all shapes) and Latin Modern mono regular are cached
\usepackage{fontspec}
\setmainfont{TeXGyreTermesX}[Extension=.otf,UprightFont=*-Regular,BoldFont=*-Bold,
  ItalicFont=*-Italic,BoldItalicFont=*-BoldItalic,SlantedFont=*-Slanted]
\setmonofont{lmmono10-regular.otf}
\setsansfont{TeXGyreTermesX}[Extension=.otf,UprightFont=*-Regular,BoldFont=*-Bold,
  ItalicFont=*-Italic,BoldItalicFont=*-BoldItalic]
\AtBeginDocument{\renewcommand{\mathbf}[1]{\text{\bfseries #1}}}
% math only at sizes whose Type-1 fonts are cached (cm 10/9/7)
\DeclareMathSizes{5}{7}{7}{7}\DeclareMathSizes{6}{7}{7}{7}\DeclareMathSizes{7}{7}{7}{7}
\DeclareMathSizes{8}{9}{7}{7}\DeclareMathSizes{9}{9}{7}{7}\DeclareMathSizes{10}{10}{7}{7}
\DeclareMathSizes{10.95}{10}{7}{7}\DeclareMathSizes{12}{10}{7}{7}\DeclareMathSizes{14.4}{10}{7}{7}
\DeclareMathSizes{17.28}{10}{7}{7}\DeclareMathSizes{20.74}{10}{7}{7}\DeclareMathSizes{24.88}{10}{7}{7}
% section headings at body size (math in headings would need cmmi12, not cached)
\renewcommand{\section}{\@startsection{section}{1}{\z@}{2.5ex plus 1ex}{1ex}{\normalfont\normalsize\bfseries}}
% figures: scale to roughly one revtex column (8.6 cm) so float placement resembles the
% two-column layout; missing files show a framed placeholder instead of failing
\AtBeginDocument{\let\preview@ig\includegraphics
  \renewcommand{\includegraphics}[2][]{\IfFileExists{#2}{\preview@ig[width=0.62\textwidth,height=0.5\textheight,keepaspectratio]{#2}}%
    {\fbox{\parbox[c][4cm][c]{0.9\columnwidth}{\centering\ttfamily [figure pending: \detokenize{#2}]}}}}}
% float parameters so that the large one-column figures can be placed near their text
\renewcommand{\topfraction}{0.95}\renewcommand{\textfraction}{0.05}\renewcommand{\floatpagefraction}{0.6}
% revtex stand-ins
\providecommand{\affiliation}[1]{\gdef\@affil{#1}}
\providecommand{\onecolumngrid}{}
\providecommand{\twocolumngrid}{}
\newenvironment{ruledtabular}{\par\noindent\hrule\smallskip}{\smallskip\hrule\par}
\newenvironment{acknowledgments}{\section*{Acknowledgments}}{}
\renewcommand{\@maketitle}{\begin{center}{\Large\bfseries\@title\par}\medskip
  {\@author\par}\smallskip{\small\@affil\par}\smallskip{\small\@date}\end{center}}
\def\@affil{}
\makeatother
\input{draft.tex}
